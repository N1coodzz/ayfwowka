from engine.decision_engine import analyze_state
from ocr.parser import parse_manual_input


def test_low_start_under():
    state = parse_manual_input("""
match=Falcons-Spirit
time=5:33
score=1-1
nw=0.5
line=54.5
under=1.94
over=1.77
""")
    decision = analyze_state(state)
    assert decision.decision == "брать"
    assert decision.best.side == "under"
